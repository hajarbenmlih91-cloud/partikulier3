/** Appel signé d'une route d'automatisation (même dérivation que AutomationHmacTrait). */
function hmacKey(secret: string) {
  const b = Buffer.from(secret, 'base64');
  if (b.length >= 32 && b.toString('base64').replace(/=+$/, '') === secret.replace(/=+$/, '')) return b;
  if (/^[a-f0-9]{64,}$/i.test(secret)) return Buffer.from(secret.slice(0, secret.length - (secret.length % 2)), 'hex');
  return Buffer.from(secret);
}
export async function automation(request: APIRequestContext, route: string, body: object,
                                 opts: { method?: string; secret?: string; badSig?: boolean } = {}) {
  const method = opts.method ?? 'POST';
  const secret = opts.secret ?? process.env.PARTIKULIER_N8N_SECRET!;
  const raw = method === 'GET' ? '' : JSON.stringify(body);
  const ts = String(Math.floor(Date.now() / 1000));
  const canonical = `${method}\n/partikulier/v1${route}\n${ts}\n${raw}`;
  let sig = crypto.createHmac('sha256', hmacKey(secret)).update(canonical).digest('hex');
  if (opts.badSig) sig = '0'.repeat(64);
  return request.fetch(`/wp-json/partikulier/v1${route}`, {
    method, data: raw || undefined,
    headers: { 'Content-Type': 'application/json', 'X-Partikulier-Automation': secret,
               'X-Partikulier-Timestamp': ts, 'X-Partikulier-Key-Id': 'env',
               'X-Partikulier-Signature': `sha256=${sig}` },
  });
}
